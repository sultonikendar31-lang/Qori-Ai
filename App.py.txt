    st.session_state.chat = genai.GenerativeModel("gemini-2.0-flash", system_instruction=SYSTEM_PROMPT).start_chat(history=[])
    st.session_state.login = False

if not st.session_state.login:
    nama = st.text_input("Nama lengkap sesuai absen:")
    if st.button("Masuk"):
        n = " ".join(nama.strip().upper().split())
        if n in DAFTAR_SISWA:
            st.session_state.login = True; st.session_state.nama = n; st.session_state.peran = DAFTAR_SISWA[n]; st.rerun()

            st.session_state.login = True; st.session_state.nama = n; st.session_state.peran = DAFTAR_DEWASA[n]; st.rerun()

            st.error("Maaf, nama kamu tidak terdaftar. Hubungi Kendar Sultoni.")
    st.stop()

st.success(f"Selamat datang, {st.session_state.nama} - {st.session_state.peran}")
for msg in st.session_state.chat.history:
    with st.chat_message("user" if msg.role=="user" else "assistant"):
        st.markdown(msg.parts[0].text)

        res = st.session_state.chat.send_message(prompt); st.markdown(res.text)
